   637→   637→function fxUpdate (dt, rawDt) {
   638→   638→  // JUICE ITEM 3 — the grid reads its brightness from CFG.juice.grid:
   639→   639→  //   baseline = max(breathing idle (c), combat intensity (d)) — user-tuned:
   640→   640→  //   combat can raise the floor but must never dim the grid below idle.
   641→   641→  //   Eased, never snaps. The whole-grid shoot flash (a) stacks on top.
   642→   642→  //   The kill pulse (b) is local (see band 03).
   643→   643→  const KG = CFG.juice.grid;
   644→   644→  const ph = 0.5 + 0.5 * Math.sin(TAU * S.time * 1000 / KG.breathPeriodMs);
   645→   645→  const breath = KG.breathMin + (KG.breathMax - KG.breathMin) * ph;
   646→   646→  if (S.mode === 'over') {
   647→   647→    // JUICE 5b — the arena holds its breath on the game-over screen: baseline
   648→   648→    // frozen (no breathing, no combat response), only the shoot flash decays.
   649→   649→  } else {
   650→   650→    let nEnemies = 0;
   651→   651→    for (const e of enemyPool.items) if (e.active && !e.dying) nEnemies++;
   652→   652→    let gTarget = breath;                          // idle floor = breathing sine
   653→   653→    if (nEnemies > 0) {
   654→   654→      const combat = KG.intensityBase + Math.min(KG.intensityMax - KG.intensityBase, nEnemies * KG.intensityScale);
   655→   655→      gTarget = Math.max(breath, combat);          // combat never dims below idle
   656→   656→    }
   657→   657→    S.gridBase += (gTarget - S.gridBase) * damp(3 / (KG.intensityEaseMs / 1000), rawDt);
   658→   658→  }
   659→   659→  S.gridShoot = Math.max(0, S.gridShoot - (KG.shootFlashAmount / (KG.shootFlashMs / 1000)) * rawDt);
   660→   660→  gridC.alpha = clamp(S.gridBase + S.gridShoot, 0, 1.6);
   661→   661→  borderC.alpha = 0.78 + 0.22 * Math.sin(S.time * 2.1);
   662→   662→
   663→   663→  // trauma → shake (decays in real time so shake survives hit-stop)
   664→   664→  S.trauma = Math.max(0, S.trauma - CFG.feel.traumaDecay * rawDt);
   665→   665→  const mag = S.trauma * S.trauma * CFG.feel.shakeMax;
   666→   666→  S.shakeX = (Math.random() * 2 - 1) * mag;
   667→   667→  S.shakeY = (Math.random() * 2 - 1) * mag * 0.85;
   668→   668→
   669→   669→  // JUICE 2.4 — additive kill shake: fixed random direction per kill, ease-out
   670→   670→  // decay, stacks to shakeMaxStack. Ticks on rawDt so the jolt keeps playing
   671→   671→  // through the hit-stop freeze (feedback survives, gameplay freezes).
   672→   672→  if (S.kMag > 0) {
   673→   673→    S.kT += rawDt;
   674→   674→    const kk = S.kT / S.kDur;
   675→   675→    if (kk >= 1) { S.kMag = 0; S.kx = 0; S.ky = 0; }
   676→   676→    else {
   677→   677→      const km = S.kMag * (1 - kk) * (1 - kk);
   678→   678→      S.kx = Math.cos(S.kAng) * km;
   679→   679→      S.ky = Math.sin(S.kAng) * km;
   680→   680→    }
   681→   681→  }
   682→   682→  worldRoot.position.set(offX + S.shakeX + S.kx, offY + S.shakeY + S.ky);
   683→   683→  worldRoot.rotation = ((Math.random() * 2 - 1) * mag) * 0.0012;
   684→   684→
   685→   685→  // fullscreen damage flash
   686→   686→  S.flashA = Math.max(0, S.flashA - 3.5 * rawDt);
   687→   687→  flashRect.alpha = S.flashA;
   688→   688→
   689→   689→  // RGB split — hurt chromatic aberration, decays fast
   690→   690→  if (rgbF) {
