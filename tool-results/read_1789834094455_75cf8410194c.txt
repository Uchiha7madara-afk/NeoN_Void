   612→// Generic fixed-size pool. Objects get .active/.visible toggles.
   613→function makePool (n, factory, layer) {
   614→  const items = [];
   615→  for (let i = 0; i < n; i++) {
   616→    const o = factory();
   617→    o.active = false; o.visible = false;
   618→    layer.addChild(o);
   619→    items.push(o);
   620→  }
   621→  return {
   622→    items,
   623→    get () {
   624→      for (const o of items) if (!o.active) { o.active = true; o.visible = true; return o; }
   625→      return null;
   626→    },
   627→    release (o) { o.active = false; o.visible = false; },
   628→    releaseAll () { for (const o of items) { o.active = false; o.visible = false; } },
   629→    countActive () { let c = 0; for (const o of items) if (o.active) c++; return c; },
   630→  };
   631→}
