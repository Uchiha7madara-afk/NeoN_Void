   493→  const mag = S.trauma * S.trauma * CFG.feel.shakeMax;

   639→// Generic fixed-size pool. Objects get .active/.visible toggles.
   640→function makePool (n, factory, layer) {
   641→  const items = [];
   642→  for (let i = 0; i < n; i++) {
   643→    const o = factory();
   644→    o.active = false; o.visible = false;
   645→    layer.addChild(o);
   646→    items.push(o);
   647→  }
   648→  return {
   649→    items,
   650→    get () {
   651→      for (const o of items) if (!o.active) { o.active = true; o.visible = true; return o; }
   652→      return null;
   653→    },
   654→    release (o) { o.active = false; o.visible = false; },
   655→    releaseAll () { for (const o of items) { o.active = false; o.visible = false; } },
   656→    countActive () { let c = 0; for (const o of items) if (o.active) c++; return c; },
   657→  };
   658→}

   663→function emit (o) {
   664→  const p = partPool.get();

   670→  p.life = p.maxLife = o.life || 0.5;
   671→  p.size0 = o.size || 1; p.grow = o.grow || 0;
   672→  p.drag = o.drag ?? 2.5;
   673→  p.alpha0 = o.alpha ?? 1;
   674→  p.alpha = p.alpha0;
   675→  p.rotation = o.rot || 0;
   676→  p.scale.set(o.size || 1);
   677→}

   704→function updateParticles (dt) {
   705→  for (const p of partPool.items) {
   706→    if (!p.active) continue;
   707→    p.life -= dt;
   708→    if (p.life <= 0) { partPool.release(p); continue; }
   709→    const k = Math.exp(-p.drag * dt);
   710→    p.vx *= k; p.vy *= k;
   711→    p.x += p.vx * dt; p.y += p.vy * dt;
   712→    const t = p.life / p.maxLife;
   713→    p.alpha = p.alpha0 * Math.min(1, t * 1.6);
   714→    const sc = p.size0 * (p.grow ? 1 + (1 - t) * p.grow : (0.4 + 0.6 * t));
   715→    p.scale.set(sc);
   716→  }
   717→}

   879→function playerUpdate (dt) {

   886→  const mag = Math.hypot(ax, ay);
   887→  P.thrust = mag > 0 ? 1 : 0;
   888→  if (mag > 0) { ax /= mag; ay /= mag; }

   892→  P.vx *= k; P.vy *= k;

   894→  if (sp > P.ship.maxSpeed) { P.vx *= P.ship.maxSpeed / sp; P.vy *= P.ship.maxSpeed / sp; }
