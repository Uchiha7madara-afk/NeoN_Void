   612→// Generic fixed-size pool. Objects get .active/.visible toggles.
   613→function makePool (n, factory, layer) {
   614→  const items = [];
   615→  for (let i = 0; i < n; i++) {
   616→    const o = factory();
   617→    o.active = false; o.visible = false;
   618→    layer.addChild(o);
   619→    items.push(o);
   620→  }
   621→  return {
   622→    items,
   623→    get () {
   624→      for (const o of items) if (!o.active) { o.active = true; o.visible = true; return o; }
   625→      return null;
   626→    },
   627→    release (o) { o.active = false; o.visible = false; },
   628→    releaseAll () { for (const o of items) { o.active = false; o.visible = false; } },
   629→    countActive () { let c = 0; for (const o of items) if (o.active) c++; return c; },
   630→  };
   631→}

   677→function updateParticles (dt) {
   678→  for (const p of partPool.items) {
   679→    if (!p.active) continue;
   680→    p.life -= dt;
   681→    if (p.life <= 0) { partPool.release(p); continue; }
   682→    const k = Math.exp(-p.drag * dt);
   683→    p.vx *= k; p.vy *= k;
   684→    p.x += p.vx * dt; p.y += p.vy * dt;
   685→    const t = p.life / p.maxLife;
   686→    p.alpha = p.alpha0 * Math.min(1, t * 1.6);
   687→    const sc = p.size0 * (p.grow ? 1 + (1 - t) * p.grow : (0.4 + 0.6 * t));
   688→    p.scale.set(sc);
   689→  }
   690→}

   801→function playerUpdate (dt) {
   802→  if (!player.alive) return;
   803→  const P = player;
   804→
   805→  // thrust in world space (WASD), inertia + drag — ship stats override defaults
   806→  let ax = (K.KeyD || K.ArrowRight ? 1 : 0) - (K.KeyA || K.ArrowLeft ? 1 : 0);
   807→  let ay = (K.KeyS || K.ArrowDown ? 1 : 0) - (K.KeyW || K.ArrowUp ? 1 : 0);
   808→  const mag = Math.hypot(ax, ay);
   809→  P.thrust = mag > 0 ? 1 : 0;
   810→  if (mag > 0) { ax /= mag; ay /= mag; }
   811→  P.vx += ax * P.ship.accel * dt;
   812→  P.vy += ay * P.ship.accel * dt;
   813→  const k = Math.exp(-P.ship.drag * dt);
   814→  P.vx *= k; P.vy *= k;
   815→  const sp = Math.hypot(P.vx, P.vy);
   816→  if (sp > P.ship.maxSpeed) { P.vx *= P.ship.maxSpeed / sp; P.vy *= P.ship.maxSpeed / sp; }
   817→  P.x += P.vx * dt; P.y += P.vy * dt;
   818→
   819→  // soft walls — bounce + squash so the box feels physical
   820→  const b = CFG.world.bound + P.ship.radius;
   821→  let bounced = 0;
   822→  if (P.x < b)     { P.x = b;     if (P.vx < 0) { P.vx *= -PC.wallBounce; bounced = 1; P.squashX = 0.16; } }
   823→  if (P.x > W - b) { P.x = W - b; if (P.vx > 0) { P.vx *= -PC.wallBounce; bounced = 1; P.squashX = 0.16; } }
   824→  if (P.y < b)     { P.y = b;     if (P.vy < 0) { P.vy *= -PC.wallBounce; bounced = 1; P.squashY = 0.16; } }
   825→  if (P.y > H - b) { P.y = H - b; if (P.vy > 0) { P.vy *= -PC.wallBounce; bounced = 1; P.squashY = 0.16; } }
   826→  if (bounced) { addTrauma(CFG.feel.traumaWall); puff(P.x, P.y, 4, 0x9d5cff, 70); }
   827→
   828→  // face the cursor (smoothed so aim stays readable)
   829→  const ta = Math.atan2(M.y - P.y, M.x - P.x);
   830→  P.angle = lerpAngle(P.angle, ta, damp(PC.turnLerp, dt));
   831→
   832→  // engine trail — rate scales with thrust and speed, color per ship
   833→  P.trailAcc += P.thrust * (0.35 + 0.65 * sp / P.ship.maxSpeed) * PC.trailRate * dt;
   834→  while (P.trailAcc >= 1) {
   835→    P.trailAcc -= 1;
   836→    const back = P.angle + Math.PI;
   837→    emit({
   838→      tex: TEX.dot, tint: P.ship.trail,
   839→      x: P.x + Math.cos(back) * 14 + rand(-3, 3),
   840→      y: P.y + Math.sin(back) * 14 + rand(-3, 3),
   841→      vx: Math.cos(back) * rand(30, 90) + P.vx * 0.25,
   842→      vy: Math.sin(back) * rand(30, 90) + P.vy * 0.25,
   843→      life: rand(0.22, 0.42), size: rand(0.45, 0.85), drag: 3.5,
   844→    });
   845→  }
   846→
   847→  // fire — hold LMB, or always when AUTO-FIRE accessibility toggle is on
   848→  P.fireCd -= dt;
   849→  const gt = CFG.gunTiers[P.gun - 1];
   850→  const rate = gt.rate * (META.upg.overcharge ? 1.18 : 1);
   851→  if ((META.autoFire || M.down) && P.fireCd <= 0) {
   852→    P.fireCd = 1 / rate;
   853→    fireBullet(gt);
   854→  }
   855→
   856→  // i-frame blink
   857→  P.iframes = Math.max(0, P.iframes - dt);
   858→  P.flashT = Math.max(0, P.flashT - dt);
   859→
   860→  // sprite transform (thrust stretch + squash + banking tilt)
   861→  P.squashX = Math.max(0, P.squashX - dt * 1.4);
   862→  P.squashY = Math.max(0, P.squashY - dt * 1.4);
   863→  const lat = Math.cos(P.angle) * P.vy - Math.sin(P.angle) * P.vx; // lateral velocity
   864→  const bank = clamp(lat * 0.0007, -0.22, 0.22);
   865→  playerSpr.texture = P.flashT > 0 ? TEX['shipW_' + P.ship.name] : TEX[shipTexName(P.ship)];
   866→  playerSpr.position.set(P.x, P.y);
   867→  playerSpr.rotation = P.angle + bank;
   868→  playerSpr.scale.set(
   869→    P.ship.scale * (1 + P.thrust * 0.10) * (1 - P.squashX * 1.6),
   870→    P.ship.scale * (1 - P.thrust * 0.06) * (1 - P.squashY * 1.6)
   871→  );
   872→  playerSpr.alpha = P.iframes > 0 ? (Math.sin(S.time * 42) > 0 ? 1 : 0.25) : 1;
   873→}

   923→function fireBullet (gt) {
   924→  const P = player;
   925→  const n = gt.count;
   926→  for (let i = 0; i < n; i++) {
   927→    const b = bulletPool.get();
   928→    if (!b) break;
   929→    let a = P.angle + rand(-PC.bulletSpread, PC.bulletSpread);
   930→    let ox = 0, oy = 0;
   931→    if (n > 1) {
   932→      if (gt.parallel) {
   933→        // TWIN — two parallel lanes
   934→        const px = -Math.sin(P.angle), py = Math.cos(P.angle);
   935→        const lane = (i - (n - 1) / 2) * 9;
   936→        ox = px * lane; oy = py * lane;
   937→        a = P.angle + rand(-PC.bulletSpread, PC.bulletSpread);
   938→      } else {
   939→        // fan spread
   940→        const t = n === 1 ? 0 : (i / (n - 1)) - 0.5;
   941→        a = P.angle + t * gt.spread * 2 + rand(-PC.bulletSpread, PC.bulletSpread);
   942→      }
   943→    }
   944→    b.texture = TEX[gt.tex];
   945→    b.x = P.x + Math.cos(P.angle) * 18 + ox;
   946→    b.y = P.y + Math.sin(P.angle) * 18 + oy;
   947→    b.vx = Math.cos(a) * gt.speed;
   948→    b.vy = Math.sin(a) * gt.speed;
   949→    b.life = gt.life;
   950→    b.dmg = gt.dmg;
   951→    b.pierce = gt.pierce;
   952→    b.last = null;
   953→    b.wob = gt.wave || 0;
   954→    b.wt = 0;
   955→    b.px = -Math.sin(a); b.py = Math.cos(a);
   956→    b.rotation = a;
   957→    b.scale.set(rand(0.9, 1.1));
   958→    // muzzle flash
   959→    emit({ tex: TEX.dot, tint: COL.core, x: b.x, y: b.y, vx: Math.cos(a) * 60, vy: Math.sin(a) * 60, life: 0.07, size: 0.9, drag: 2 });
   960→  }
   961→  A.shoot(P.gun);
   962→}

  1098→function updateEnemies (dt) {
  1099→  const chasing = player.alive && S.mode === 'playing';
  1100→  const actives = [];
  1101→  for (const d of enemyPool.items) if (d.active) actives.push(d);
  1102→
  1103→  for (const d of actives) {
  1104→    d.flashT = Math.max(0, d.flashT - dt);
  1105→    d.spawnK = Math.min(1, d.spawnK + dt * 4.5);
  1106→    const c = d.cfg;
  1107→
  1108→    if (d.kind === 'drone') {
  1109→      d.rotA += c.spin * dt;
  1110→      if (chasing) {
  1111→        const dx = player.x - d.x, dy = player.y - d.y;
  1112→        const wob = Math.sin(S.time * c.weaveFreq + d.phase) * c.weaveAmp;
  1113→        const ang = Math.atan2(dy, dx) + wob;
  1114→        const ms = c.baseSpeed * Math.min(c.speedCap, 1 + (S.wave - 1) * c.speedRamp) * DIFF().speed;
  1115→        const dvx = Math.cos(ang) * ms - d.vx, dvy = Math.sin(ang) * ms - d.vy;
  1116→        const dl = Math.hypot(dvx, dvy) || 1;
  1117→        const st = Math.min(c.accel * dt, dl);
  1118→        d.vx += (dvx / dl) * st; d.vy += (dvy / dl) * st;
  1119→        // separation — keeps the swarm readable instead of one blob
  1120→        for (const o of actives) {
  1121→          if (o === d || o.kind !== 'drone') continue;
  1122→          const sx = d.x - o.x, sy = d.y - o.y;
  1123→          const sd = Math.hypot(sx, sy);
  1124→          if (sd > 0.01 && sd < c.sepDist) {
  1125→            d.vx += (sx / sd) * c.sepForce * dt;
  1126→            d.vy += (sy / sd) * c.sepForce * dt;
  1127→          }
  1128→        }
  1129→      } else { d.vx *= Math.exp(-0.8 * dt); d.vy *= Math.exp(-0.8 * dt); }
  1130→      d.texture = d.flashT > 0 ? TEX.droneW : TEX.drone;
  1131→      d.rotation = d.rotA;
  1132→      const pulse = 1 + Math.sin(S.time * c.pulseRate + d.phase) * c.pulseAmp;
  1133→      const popIn = 1 - Math.pow(1 - d.spawnK, 3);
  1134→      d.scale.set(popIn * pulse * (d.spawnK < 1 ? 1.15 : 1));
  1135→    }
  1136→
  1137→    else if (d.kind === 'hunter') {
  1138→      if (chasing) {
  1139→        const dx = player.x - d.x, dy = player.y - d.y;
  1140→        const dist = Math.hypot(dx, dy) || 1;
  1141→        const ms = c.baseSpeed * sharedSpeedMult();
  1142→        let tx, ty;
  1143→        if (dist > c.ring + c.ringSlack)      { tx = dx / dist; ty = dy / dist; }
  1144→        else if (dist < c.ring - c.ringSlack) { tx = -dx / dist; ty = -dy / dist; }
  1145→        else { tx = (-dy / dist) * d.orb; ty = (dx / dist) * d.orb; }
  1146→        const dvx = tx * ms - d.vx, dvy = ty * ms - d.vy;
  1147→        const dl = Math.hypot(dvx, dvy) || 1;
  1148→        const st = Math.min(c.accel * dt, dl);
  1149→        d.vx += (dvx / dl) * st; d.vy += (dvy / dl) * st;
  1150→
  1151→        // fire cycle: charge telegraph → predictive shot
  1152→        d.fireT -= dt;
  1153→        if (d.fireT <= 0 && d.spawnK >= 1) {
  1154→          const predict = Math.min(0.95, c.predict + S.wave * 0.02);
  1155→          const tof = dist / (c.bulletSpeed * DIFF().bullet);

  1450→function buildQueue (budget, wave) {
  1451→  const avail = ['drone'];
  1452→  if (wave >= CFG.waves.unlock.hunter)  avail.push('hunter');
  1453→  if (wave >= CFG.waves.unlock.spinner) avail.push('spinner');
  1454→  if (wave >= CFG.waves.unlock.tank)    avail.push('tank');
  1455→  const q = [];
  1456→  let b = budget, guard = 0;
  1457→  while (b > 0 && guard++ < 500) {
  1458→    // weighted pick among affordable types
  1459→    let total = 0;
  1460→    const aff = [];
  1461→    for (const k of avail) {
  1462→      if (CFG.waves.cost[k] <= b) { aff.push(k); total += CFG.waves.weights[k]; }
  1463→    }
  1464→    if (!aff.length) { q.push('drone'); b -= 1; continue; }
  1465→    let r = Math.random() * total;
  1466→    let pick = aff[0];
  1467→    for (const k of aff) { r -= CFG.waves.weights[k]; if (r <= 0) { pick = k; break; } }
  1468→    q.push(pick);
  1469→    b -= CFG.waves.cost[pick];
  1470→  }
  1471→  // shuffle lightly so bigs don't always come in the same order
  1472→  for (let i = q.length - 1; i > 0; i--) {
  1473→    if (Math.random() < 0.25) {
  1474→      const j = (Math.random() * (i + 1)) | 0;
  1475→      [q[i], q[j]] = [q[j], q[i]];
  1476→    }
  1477→  }
  1478→  return q;
  1479→}

  1775→/* ===== 11 AUDIO — Web Audio synth, zero files ===== */
  1776→const A = (() => {
  1777→  let ctx = null, master = null, noiseBuf = null;
  1778→  let muted = !!META.muted;
  1779→  function init () {
  1780→    if (ctx) return;
  1781→    const AC = window.AudioContext || window.webkitAudioContext;
  1782→    if (!AC) return;
  1783→    ctx = new AC();
  1784→    master = ctx.createGain();
  1785→    master.gain.value = muted ? 0 : CFG.audio.master;
  1786→    master.connect(ctx.destination);
  1787→    noiseBuf = ctx.createBuffer(1, ctx.sampleRate * 0.5 | 0, ctx.sampleRate);
  1788→    const d = noiseBuf.getChannelData(0);
  1789→    for (let i = 0; i < d.length; i++) d[i] = Math.random() * 2 - 1;
  1790→  }
  1791→  function resume () { if (ctx && ctx.state === 'suspended') ctx.resume(); }
  1792→  function tone ({ type, f0, f1, dur, vol, at }) {
  1793→    if (!ctx || muted) return;
  1794→    const t = ctx.currentTime + (at || 0);
  1795→    const o = ctx.createOscillator(), g = ctx.createGain();
  1796→    o.type = type;
  1797→    o.frequency.setValueAtTime(Math.max(1, f0), t);
  1798→    o.frequency.exponentialRampToValueAtTime(Math.max(1, f1), t + dur);
  1799→    g.gain.setValueAtTime(vol, t);
  1800→    g.gain.exponentialRampToValueAtTime(0.0008, t + dur);
  1801→    o.connect(g); g.connect(master);
  1802→    o.start(t); o.stop(t + dur + 0.03);
  1803→  }
  1804→  function noise ({ dur, vol, lp0, lp1, at }) {
  1805→    if (!ctx || muted) return;
  1806→    const t = ctx.currentTime + (at || 0);
  1807→    const src = ctx.createBufferSource(); src.buffer = noiseBuf; src.loop = true;
  1808→    const f = ctx.createBiquadFilter(); f.type = 'lowpass';
  1809→    f.frequency.setValueAtTime(Math.max(40, lp0), t);
  1810→    f.frequency.exponentialRampToValueAtTime(Math.max(40, lp1), t + dur);
  1811→    const g = ctx.createGain();
  1812→    g.gain.setValueAtTime(vol, t);
  1813→    g.gain.exponentialRampToValueAtTime(0.0008, t + dur);
  1814→    src.connect(f); f.connect(g); g.connect(master);
  1815→    src.start(t); src.stop(t + dur + 0.03);
  1816→  }
  1817→  return {
  1818→    init, resume,
  1819→    shoot (gun) {
  1820→      const c = CFG.audio.shoot;
  1821→      tone({ type: c.type, f0: c.f0 * (1 + gun * 0.07) * rand(0.94, 1.06), f1: c.f0 * c.fMul, dur: c.dur, vol: c.vol });
  1822→    },
  1823→    kill ()  { const c = CFG.audio.kill; noise(c.noise); tone(c.tone); },
  1824→    hurt ()  { const c = CFG.audio.hurt; tone(c.tone); noise(c.noise); },
  1825→    coin ()  { for (const t of CFG.audio.coin.tones) tone(t); },
  1826→    gun ()   { const c = CFG.audio.gun; tone(c.tone); noise(c.noise); },
  1827→    heal ()  { for (const t of CFG.audio.heal.tones) tone(t); },
  1828→    ui ()    { tone(CFG.audio.ui.tone); },
  1829→    boom ()  { const c = CFG.audio.boom; noise(c.noise); tone(c.tone); },
  1830→    tier ()  { for (const t of CFG.audio.tier.tones) tone(t); },
  1831→    toggleMute () {
  1832→      muted = !muted;
  1833→      META.muted = muted; saveMeta();
  1834→      if (master) master.gain.value = muted ? 0 : CFG.audio.master;
  1835→      return muted;
  1836→    },
  1837→    setMuted (m) { muted = !!m; if (master) master.gain.value = muted ? 0 : CFG.audio.master; },
  1838→    get muted () { return muted; },
  1839→  };
  1840→})();
