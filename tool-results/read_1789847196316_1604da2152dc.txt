   584→function fxUpdate (dt, rawDt) {
   585→  // JUICE ITEM 3 — the grid reads its brightness from CFG.juice.grid:
   586→  //   baseline = max(breathing idle (c), combat intensity (d)) — user-tuned:
   587→  //   combat can raise the floor but must never dim the grid below idle.
   588→  //   Eased, never snaps. The whole-grid shoot flash (a) stacks on top.
   589→  //   The kill pulse (b) is local (see band 03).
   590→  const KG = CFG.juice.grid;
   591→  const ph = 0.5 + 0.5 * Math.sin(TAU * S.time * 1000 / KG.breathPeriodMs);
   592→  const breath = KG.breathMin + (KG.breathMax - KG.breathMin) * ph;
   593→  if (S.mode === 'over') {
   594→    // JUICE 5b — the arena holds its breath on the game-over screen: baseline
   595→    // frozen (no breathing, no combat response), only the shoot flash decays.
   596→  } else {
   597→    let nEnemies = 0;
   598→    for (const e of enemyPool.items) if (e.active && !e.dying) nEnemies++;
   599→    let gTarget = breath;                          // idle floor = breathing sine
   600→    if (nEnemies > 0) {
   601→      const combat = KG.intensityBase + Math.min(KG.intensityMax - KG.intensityBase, nEnemies * KG.intensityScale);
   602→      gTarget = Math.max(breath, combat);          // combat never dims below idle
   603→    }
   604→    S.gridBase += (gTarget - S.gridBase) * damp(3 / (KG.intensityEaseMs / 1000), rawDt);
   605→  }
   606→  S.gridShoot = Math.max(0, S.gridShoot - (KG.shootFlashAmount / (KG.shootFlashMs / 1000)) * rawDt);
   607→  gridC.alpha = clamp(S.gridBase + S.gridShoot, 0, 1.6);
   608→  borderC.alpha = 0.78 + 0.22 * Math.sin(S.time * 2.1);
   609→
   610→  // trauma → shake (decays in real time so shake survives hit-stop)
   611→  S.trauma = Math.max(0, S.trauma - CFG.feel.traumaDecay * rawDt);
   612→  const mag = S.trauma * S.trauma * CFG.feel.shakeMax;
   613→  S.shakeX = (Math.random() * 2 - 1) * mag;
   614→  S.shakeY = (Math.random() * 2 - 1) * mag * 0.85;
   615→
   616→  // JUICE 2.4 — additive kill shake: fixed random direction per kill, ease-out
   617→  // decay, stacks to shakeMaxStack. Ticks on rawDt so the jolt keeps playing
   618→  // through the hit-stop freeze (feedback survives, gameplay freezes).
   619→  if (S.kMag > 0) {
   620→    S.kT += rawDt;
   621→    const kk = S.kT / S.kDur;
   622→    if (kk >= 1) { S.kMag = 0; S.kx = 0; S.ky = 0; }
   623→    else {
   624→      const km = S.kMag * (1 - kk) * (1 - kk);
   625→      S.kx = Math.cos(S.kAng) * km;
   626→      S.ky = Math.sin(S.kAng) * km;
   627→    }
   628→  }
   629→  worldRoot.position.set(offX + S.shakeX + S.kx, offY + S.shakeY + S.ky);
   630→  worldRoot.rotation = ((Math.random() * 2 - 1) * mag) * 0.0012;
   631→
   632→  // fullscreen damage flash
   633→  S.flashA = Math.max(0, S.flashA - 3.5 * rawDt);
   634→  flashRect.alpha = S.flashA;
   635→
   636→  // RGB split — hurt chromatic aberration, decays fast
   637→  if (rgbF) {
   638→    S.rgbT = Math.max(0, S.rgbT - rawDt);
   639→    const k = (S.rgbT / CFG.feel.rgbHurt) * CFG.feel.rgbMag;
   640→    rgbF.red = [k, 0];
   641→    rgbF.green = [0, k * 0.6];
   642→    rgbF.blue = [-k * 0.5, 0];
   643→  }
   644→
   645→  // JUICE 5a — red edge vignette decays linearly over vignetteMs (real time)
   646→  const DKv = CFG.juice.death;
   647→  S.vigA = Math.max(0, S.vigA - (DKv.vignetteAlpha / Math.max(0.001, DKv.vignetteMs / 1000)) * rawDt);
   648→  vignetteSpr.alpha = S.vigA;
