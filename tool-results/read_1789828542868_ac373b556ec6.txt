   318→// Generic fixed-size pool. Objects get .active/.visible toggles.
   319→function makePool (n, factory, layer) {
   320→  const items = [];
   321→  for (let i = 0; i < n; i++) {
   322→    const o = factory();
   323→    o.active = false; o.visible = false;
   324→    layer.addChild(o);
   325→    items.push(o);
   326→  }
   327→  return {
   328→    items,
   329→    get () {
   330→      for (const o of items) if (!o.active) { o.active = true; o.visible = true; return o; }
   331→      return null;
   332→    },
   333→    release (o) { o.active = false; o.visible = false; },
   334→    releaseAll () { for (const o of items) { o.active = false; o.visible = false; } },
   335→    countActive () { let c = 0; for (const o of items) if (o.active) c++; return c; },
   336→  };
   337→}

   383→function updateParticles (dt) {
   384→  for (const p of partPool.items) {
   385→    if (!p.active) continue;
   386→    p.life -= dt;
   387→    if (p.life <= 0) { partPool.release(p); continue; }
   388→    const k = Math.exp(-p.drag * dt);
   389→    p.vx *= k; p.vy *= k;
   390→    p.x += p.vx * dt; p.y += p.vy * dt;
   391→    const t = p.life / p.maxLife;
   392→    p.alpha = p.alpha0 * Math.min(1, t * 1.6);
   393→    const sc = p.size0 * (p.grow ? 1 + (1 - t) * p.grow : (0.4 + 0.6 * t));
   394→    p.scale.set(sc);
   395→  }
   396→}

   494→function playerUpdate (dt) {
   495→  if (!player.alive) return;
   496→  const P = player;
   497→
   498→  // thrust in world space (WASD), inertia + drag
   499→  let ax = (K.KeyD || K.ArrowRight ? 1 : 0) - (K.KeyA || K.ArrowLeft ? 1 : 0);
   500→  let ay = (K.KeyS || K.ArrowDown ? 1 : 0) - (K.KeyW || K.ArrowUp ? 1 : 0);
   501→  const mag = Math.hypot(ax, ay);
   502→  P.thrust = mag > 0 ? 1 : 0;
   503→  if (mag > 0) { ax /= mag; ay /= mag; }
   504→  P.vx += ax * PC.accel * dt;
   505→  P.vy += ay * PC.accel * dt;
   506→  const k = Math.exp(-PC.drag * dt);
   507→  P.vx *= k; P.vy *= k;
   508→  const sp = Math.hypot(P.vx, P.vy);
   509→  if (sp > PC.maxSpeed) { P.vx *= PC.maxSpeed / sp; P.vy *= PC.maxSpeed / sp; }
   510→  P.x += P.vx * dt; P.y += P.vy * dt;
   511→
   512→  // soft walls — bounce + squash so the box feels physical
   513→  const b = CFG.world.bound + PC.radius;
   514→  let bounced = 0;
   515→  if (P.x < b)     { P.x = b;     if (P.vx < 0) { P.vx *= -PC.wallBounce; bounced = 1; P.squashX = 0.16; } }
   516→  if (P.x > W - b) { P.x = W - b; if (P.vx > 0) { P.vx *= -PC.wallBounce; bounced = 1; P.squashX = 0.16; } }
   517→  if (P.y < b)     { P.y = b;     if (P.vy < 0) { P.vy *= -PC.wallBounce; bounced = 1; P.squashY = 0.16; } }
   518→  if (P.y > H - b) { P.y = H - b; if (P.vy > 0) { P.vy *= -PC.wallBounce; bounced = 1; P.squashY = 0.16; } }
   519→  if (bounced) { addTrauma(CFG.feel.traumaWall); puff(P.x, P.y, 4, 0x9d5cff, 70); }
   520→
   521→  // face the cursor (smoothed so aim stays readable)
   522→  const ta = Math.atan2(M.y - P.y, M.x - P.x);
   523→  P.angle = lerpAngle(P.angle, ta, damp(PC.turnLerp, dt));
   524→
   525→  // pink engine trail — rate scales with thrust and speed
   526→  P.trailAcc += P.thrust * (0.35 + 0.65 * sp / PC.maxSpeed) * PC.trailRate * dt;
   527→  while (P.trailAcc >= 1) {
   528→    P.trailAcc -= 1;
   529→    const back = P.angle + Math.PI;
   530→    emit({
   531→      tex: TEX.dot, tint: COL.trail,
   532→      x: P.x + Math.cos(back) * 14 + rand(-3, 3),
   533→      y: P.y + Math.sin(back) * 14 + rand(-3, 3),
   534→      vx: Math.cos(back) * rand(30, 90) + P.vx * 0.25,
   535→      vy: Math.sin(back) * rand(30, 90) + P.vy * 0.25,
   536→      life: rand(0.22, 0.42), size: rand(0.45, 0.85), drag: 3.5,
   537→    });
   538→  }
   539→
   540→  // fire
   541→  P.fireCd -= dt;
   542→  if (M.down && P.fireCd <= 0) {
   543→    P.fireCd = 1 / PC.fireRate;
   544→    fireBullet();
   545→  }
   546→
   547→  // i-frame blink
   548→  P.iframes = Math.max(0, P.iframes - dt);
   549→  P.flashT = Math.max(0, P.flashT - dt);
   550→
   551→  // sprite transform (thrust stretch + squash)
   552→  P.squashX = Math.max(0, P.squashX - dt * 1.4);
   553→  P.squashY = Math.max(0, P.squashY - dt * 1.4);
   554→  playerSpr.texture = P.flashT > 0 ? TEX.shipW : TEX.ship;
   555→  playerSpr.position.set(P.x, P.y);
   556→  playerSpr.rotation = P.angle;
   557→  playerSpr.scale.set((1 + P.thrust * 0.10) * (1 - P.squashX * 1.6), (1 - P.thrust * 0.06) * (1 - P.squashY * 1.6));
   558→  playerSpr.alpha = P.iframes > 0 ? (Math.sin(S.time * 42) > 0 ? 1 : 0.25) : 1;
   559→}

   699→function droneMaxSpeed (d) { return droneSpeedFor(Math.max(1, S.wave)); }

   701→function updateDrones (dt) {
   702→  const chasing = player.alive && (S.mode === 'playing');
   703→  const actives = [];
   704→  for (const d of dronePool.items) if (d.active) actives.push(d);
   705→
   706→  for (const d of actives) {
   707→    d.rotA += DC.spin * dt;
   708→    d.flashT = Math.max(0, d.flashT - dt);
   709→    d.spawnK = Math.min(1, d.spawnK + dt * 4.5);
   710→
   711→    if (chasing) {
   712→      // steer toward player with a sinusoidal weave so groups don't look robotic
   713→      const dx = player.x - d.x, dy = player.y - d.y;
   714→      const dist = Math.hypot(dx, dy) || 1;
   715→      const wob = Math.sin(S.time * DC.weaveFreq + d.phase) * DC.weaveAmp;
   716→      const ang = Math.atan2(dy, dx) + wob;
   717→      const ms = droneMaxSpeed(d);
   718→      const dvx = Math.cos(ang) * ms - d.vx;
   719→      const dvy = Math.sin(ang) * ms - d.vy;
   720→      const dl = Math.hypot(dvx, dvy) || 1;
   721→      const st = Math.min(DC.accel * dt, dl);
   722→      d.vx += (dvx / dl) * st; d.vy += (dvy / dl) * st;
   723→
   724→      // separation — keeps the swarm readable instead of one blob
   725→      for (const o of actives) {
   726→        if (o === d) continue;
   727→        const sx = d.x - o.x, sy = d.y - o.y;
   728→        const sd = Math.hypot(sx, sy);
   729→        if (sd > 0.01 && sd < DC.sepDist) {
   730→          d.vx += (sx / sd) * DC.sepForce * dt;
   731→          d.vy += (sy / sd) * DC.sepForce * dt;
   732→        }
   733→      }
   734→    } else {
   735→      d.vx *= Math.exp(-0.8 * dt); d.vy *= Math.exp(-0.8 * dt);
   736→    }
   737→
   738→    d.x += d.vx * dt; d.y += d.vy * dt;
   739→
   740→    // soft walls
   741→    const m = CFG.world.bound + 8;
   742→    if (d.x < m)     { d.x = m;     if (d.vx < 0) d.vx *= -0.5; }
   743→    if (d.x > W - m) { d.x = W - m; if (d.vx > 0) d.vx *= -0.5; }
   744→    if (d.y < m)     { d.y = m;     if (d.vy < 0) d.vy *= -0.5; }
   745→    if (d.y > H - m) { d.y = H - m; if (d.vy > 0) d.vy *= -0.5; }
   746→
   747→    // visuals
   748→    d.texture = d.flashT > 0 ? TEX.droneW : TEX.drone;
   749→    d.rotation = d.rotA;
   750→    const pulse = 1 + Math.sin(S.time * DC.pulseRate + d.phase) * DC.pulseAmp;
   751→    const popIn = 1 - Math.pow(1 - d.spawnK, 3); // ease-out back-ish
   752→    d.scale.set(popIn * pulse * (d.spawnK < 1 ? 1.15 : 1));
   753→  }
   754→
   755→  // contact damage
   756→  if (chasing && player.iframes <= 0) {
   757→    for (const d of actives) {
   758→      const rr = PC.radius + DC.radius;
   759→      if (Math.abs(d.x - player.x) < rr && Math.abs(d.y - player.y) < rr) {
   760→        const dist = Math.hypot(d.x - player.x, d.y - player.y);
   761→        if (dist < rr) {
   762→          killDrone(d, false);          // the kamikaze drone pops too — no score
   763→          playerHit(player.x - d.x, player.y - d.y);
   764→          break;
   765→        }
   766→      }
   767→    }
   768→  }
   769→}

   997→/* ===== 11 AUDIO — Web Audio synth, zero files ===== */
   998→const A = (() => {
   999→  let ctx = null, master = null, noiseBuf = null, muted = false;
  1000→  function init () {
  1001→    if (ctx) return;
  1002→    const AC = window.AudioContext || window.webkitAudioContext;
  1003→    if (!AC) return;
  1004→    ctx = new AC();
  1005→    master = ctx.createGain();
  1006→    master.gain.value = CFG.audio.master;
  1007→    master.connect(ctx.destination);
  1008→    noiseBuf = ctx.createBuffer(1, ctx.sampleRate * 0.5 | 0, ctx.sampleRate);
  1009→    const d = noiseBuf.getChannelData(0);
  1010→    for (let i = 0; i < d.length; i++) d[i] = Math.random() * 2 - 1;
  1011→  }
  1012→  function resume () { if (ctx && ctx.state === 'suspended') ctx.resume(); }
  1013→  function tone ({ type, f0, f1, dur, vol }) {
  1014→    if (!ctx || muted) return;
  1015→    const t = ctx.currentTime;
  1016→    const o = ctx.createOscillator(), g = ctx.createGain();
  1017→    o.type = type;
  1018→    o.frequency.setValueAtTime(Math.max(1, f0), t);
  1019→    o.frequency.exponentialRampToValueAtTime(Math.max(1, f1), t + dur);
  1020→    g.gain.setValueAtTime(vol, t);
  1021→    g.gain.exponentialRampToValueAtTime(0.0008, t + dur);
  1022→    o.connect(g); g.connect(master);
  1023→    o.start(t); o.stop(t + dur + 0.03);
  1024→  }
  1025→  function noise ({ dur, vol, lp0, lp1 }) {
  1026→    if (!ctx || muted) return;
  1027→    const t = ctx.currentTime;
  1028→    const src = ctx.createBufferSource(); src.buffer = noiseBuf; src.loop = true;
  1029→    const f = ctx.createBiquadFilter(); f.type = 'lowpass';
  1030→    f.frequency.setValueAtTime(Math.max(40, lp0), t);
  1031→    f.frequency.exponentialRampToValueAtTime(Math.max(40, lp1), t + dur);
  1032→    const g = ctx.createGain();
  1033→    g.gain.setValueAtTime(vol, t);
  1034→    g.gain.exponentialRampToValueAtTime(0.0008, t + dur);
  1035→    src.connect(f); f.connect(g); g.connect(master);
  1036→    src.start(t); src.stop(t + dur + 0.03);
  1037→  }
  1038→  return {
  1039→    init, resume,
  1040→    shoot () { const c = CFG.audio.shoot; tone({ type: c.type, f0: c.f0 * rand(0.94, 1.06), f1: c.f0 * c.fMul, dur: c.dur, vol: c.vol }); },
  1041→    kill ()  { const c = CFG.audio.kill; noise(c.noise); tone(c.tone); },
  1042→    hurt ()  { const c = CFG.audio.hurt; tone(c.tone); noise(c.noise); },
  1043→    toggleMute () { muted = !muted; if (master) master.gain.value = muted ? 0 : CFG.audio.master; return muted; },
  1044→  };
  1045→})();
