/* ===== 04 PLAYER — stats come from the equipped ship ===== */ const PC = CFG.player; const player = { x: W / 2, y: H / 2, vx: 0, vy: 0, angle: -Math.PI / 2, // facing (lerped toward mouse) hp: 3, hpMax: 3, iframes: 0, fireCd: 0, gun: 1, alive: true, thrust: 0, trailAcc: 0, squashX: 0, squashY: 0, // wall-bounce squash timers flashT: 0, ship: CFG.ships.vector, }; const playerSpr = new PIXI.Sprite(TEX.ship); playerSpr.anchor.set(0.5); glow.addChild(playerSpr); function shipTexName (ship) { return 'ship_' + ship.name; } function playerReset () { const ship = CFG.ships[META.ship] || CFG.ships.vector; const D = DIFF(); player.ship = ship; player.x = W / 2; player.y = H / 2; player.vx = player.vy = 0; player.angle = -Math.PI / 2; player.hpMax = ship.hp + D.hpBonus; player.hp = player.hpMax; player.iframes = 0; player.fireCd = 0; player.gun = META.upg.twin ? 2 : 1; player.alive = true; player.flashT = 0; player.trailAcc = 0; playerSpr.texture = TEX[shipTexName(ship)]; playerSpr.visible = true; S.gun = player.gun; } function playerUpdate (dt) { if (!player.alive) return; const P = player; // thrust in world space (WASD), inertia + drag — ship stats override defaults let ax = (K.KeyD || K.ArrowRight ? 1 : 0) - (K.KeyA || K.ArrowLeft ? 1 : 0); let ay = (K.KeyS || K.ArrowDown ? 1 : 0) - (K.KeyW || K.ArrowUp ? 1 : 0); const mag = Math.hypot(ax, ay); P.thrust = mag > 0 ? 1 : 0; if (mag > 0) { ax /= mag; ay /= mag; } P.vx += ax * P.ship.accel * dt; P.vy += ay * P.ship.accel * dt; const k = Math.exp(-P.ship.drag * dt); P.vx *= k; P.vy *= k; const sp = Math.hypot(P.vx, P.vy); if (sp > P.ship.maxSpeed) { P.vx *= P.ship.maxSpeed / sp; P.vy *= P.ship.maxSpeed / sp; } P.x += P.vx * dt; P.y += P.vy * dt; // soft walls — bounce + squash so the box feels physical const b = CFG.world.bound + P.ship.radius; let bounced = 0; if (P.x < b) { P.x = b; if (P.vx < 0) { P.vx *= -PC.wallBounce; bounced = 1; P.squashX = 0.16; } } if (P.x > W - b) { P.x = W - b; if (P.vx > 0) { P.vx *= -PC.wallBounce; bounced = 1; P.squashX = 0.16; } } if (P.y < b) { P.y = b; if (P.vy < 0) { P.vy *= -PC.wallBounce; bounced = 1; P.squashY = 0.16; } } if (P.y > H - b) { P.y = H - b; if (P.vy > 0) { P.vy *= -PC.wallBounce; bounced = 1; P.squashY = 0.16; } } if (bounced) { addTrauma(CFG.feel.traumaWall); puff(P.x, P.y, 4, 0x9d5cff, 70); } // face the cursor (smoothed so aim stays readable) const ta = Math.atan2(M.y - P.y, M.x - P.x); P.angle = lerpAngle(P.angle, ta, damp(PC.turnLerp, dt)); // engine trail — rate scales with thrust and speed, color per ship P.trailAcc += P.thrust * (0.35 + 0.65 * sp / P.ship.maxSpeed) * PC.trailRate * dt; while (P.trailAcc >= 1) { P.trailAcc -= 1; const back = P.angle + Math.PI; emit({ tex: TEX.dot, tint: P.ship.trail, x: P.x + Math.cos(back) * 14 + rand(-3, 3), y: P.y + Math.sin(back) * 14 + rand(-3, 3), vx: Math.cos(back) * rand(30, 90) + P.vx * 0.25, vy: Math.sin(back) * rand(30, 90) + P.vy * 0.25, life: rand(0.22, 0.42), size: rand(0.45, 0.85), drag: 3.5, }); } // fire — hold LMB, or always when AUTO-FIRE accessibility toggle is on P.fireCd -= dt; const gt = CFG.gunTiers[P.gun - 1]; const rate = gt.rate * (META.upg.overcharge ? 1.18 : 1); if ((META.autoFire || M.down) && P.fireCd <= 0) { P.fireCd = 1 / rate; fireBullet(gt); } // i-frame blink P.iframes = Math.max(0, P.iframes - dt); P.flashT = Math.max(0, P.flashT - dt); // sprite transform (thrust stretch + squash + banking tilt) P.squashX = Math.max(0, P.squashX - dt * 1.4); P.squashY = Math.max(0, P.squashY - dt * 1.4); const lat = Math.cos(P.angle) * P.vy - Math.sin(P.angle) * P.vx; // lateral velocity const bank = clamp(lat * 0.0007, -0.22, 0.22); playerSpr.texture = P.flashT > 0 ? TEX['shipW_' + P.ship.name] : TEX[shipTexName(P.ship)]; playerSpr.position.set(P.x, P.y); playerSpr.rotation = P.angle + bank; playerSpr.scale.set( P.ship.scale * (1 + P.thrust * 0.10) * (1 - P.squashX * 1.6), P.ship.scale * (1 - P.thrust * 0.06) * (1 - P.squashY * 1.6) ); playerSpr.alpha = P.iframes > 0 ? (Math.sin(S.time * 42) > 0 ? 1 : 0.25) : 1; } function playerHit (dx, dy, knockMul = 1) { const P = player; P.hp -= 1; P.iframes = DIFF().iframes; S.flashA = CFG.feel.hurtFlash; S.rgbT = CFG.feel.rgbHurt; addTrauma(CFG.feel.traumaHurt); hitstop(CFG.feel.hitstopHurt); // knockback away from the source const d = Math.hypot(dx, dy) || 1; P.vx += (dx / d) * 220 * knockMul; P.vy += (dy / d) * 220 * knockMul; puff(P.x, P.y, 10, P.ship.color, 160); A.hurt(); if (S.combo > 0) { S.combo = 0; S.comboT = 0; popup(P.x, P.y - 26, 'COMBO LOST', '#8f7fb8', 12); } hudHP(); if (P.hp <= 0) playerDie(); } function playerDie () { const P = player; P.alive = false; playerSpr.visible = false; M.down = false; // big ship-colored detonation for (let i = 0; i < 26; i++) { const a = rand(TAU), sp = rand(120, 520); emit({ tex: TEX.spark, tint: Math.random() < 0.5 ? P.ship.color : P.ship.trail, x: P.x, y: P.y, vx: Math.cos(a) * sp, vy: Math.sin(a) * sp, life: rand(0.4, 0.9), size: rand(0.6, 1.2), drag: 2.4, rot: rand(TAU) }); } for (let i = 0; i < 6; i++) { const a = rand(TAU), sp = rand(20, 110); emit({ tex: TEX.dot, tint: P.ship.color, x: P.x, y: P.y, vx: Math.cos(a) * sp, vy: Math.sin(a) * sp, life: rand(0.6, 1.1), size: rand(1.2, 2), grow: 1.6, drag: 1.4, alpha: 0.6 }); } ring(P.x, P.y, P.ship.color, 3.2, 0.45); ring(P.x, P.y, P.ship.trail, 2.2, 0.6); addTrauma(1); hitstop(0.09); S.flashA = 0.32; A.boom(); S.mode = 'dying'; S.dyingT = CFG.feel.dyingTime; S.timescale = CFG.feel.dyingSlowmo; } /* ===== 05 BULLETS — player gun tiers 1..5 ===== */ const bulletPool = makePool(CFG.pools.bullets, () => spriteOf(TEX.bullet), bulletL); function fireBullet (gt) { const P = player; const n = gt.count; for (let i = 0; i < n; i++) { const b = bulletPool.get(); if (!b) break; let a = P.angle + rand(-PC.bulletSpread, PC.bulletSpread); let ox = 0, oy = 0; if (n > 1) { if (gt.parallel) { // TWIN — two parallel lanes const px = -Math.sin(P.angle), py = Math.cos(P.angle); const lane = (i - (n - 1) / 2) * 9; ox = px * lane; oy = py * lane; a = P.angle + rand(-PC.bulletSpread, PC.bulletSpread); } else { // fan spread const t = n === 1 ? 0 : (i / (n - 1)) - 0.5; a = P.angle + t * gt.spread * 2 + rand(-PC.bulletSpread, PC.bulletSpread); } } b.texture = TEX[gt.tex]; b.x = P.x + Math.cos(P.angle) * 18 + ox; b.y = P.y + Math.sin(P.angle) * 18 + oy; b.vx = Math.cos(a) * gt.speed; b.vy = Math.sin(a) * gt.speed; b.life = gt.life; b.dmg = gt.dmg; b.pierce = gt.pierce; b.last = null; b.wob = gt.wave || 0; b.wt = 0; b.px = -Math.sin(a); b.py = Math.cos(a); b.rotation = a; b.scale.set(rand(0.9, 1.1)); // muzzle flash emit({ tex: TEX.dot, tint: COL.core, x: b.x, y: b.y, vx: Math.cos(a) * 60, vy: Math.sin(a) * 60, life: 0.07, size: 0.9, drag: 2 }); } A.shoot(P.gun); } function updateBullets (dt) { for (const b of bulletPool.items) { if (!b.active) continue; b.life -= dt; if (b.life <= 0 || b.x < -20 || b.x > W + 20 || b.y < -20 || b.y > H + 20) { bulletPool.release(b); continue; } b.wt += dt; const wv = b.wob ? Math.cos(b.wt * 16) * b.wob * 420 : 0; b.x += (b.vx + b.px * wv) * dt; b.y += (b.vy + b.py * wv) * dt; } }
